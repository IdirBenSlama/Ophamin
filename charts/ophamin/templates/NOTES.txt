{{- if .Values.ingress.enabled }}
{{- range .Values.ingress.hosts }}
The HTTP API is exposed via Ingress at:
{{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}{{ .path }}
{{- end }}

{{- else if contains "NodePort" .Values.service.type }}
Get the NodePort:
  export NODE_PORT=$(kubectl get -n {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "ophamin.fullname" . }}-http)
  export NODE_IP=$(kubectl get nodes -n {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "Ophamin HTTP API: http://$NODE_IP:$NODE_PORT"

{{- else if contains "LoadBalancer" .Values.service.type }}
Wait for the LoadBalancer external IP:
  kubectl get svc -n {{ .Release.Namespace }} {{ include "ophamin.fullname" . }}-http --watch

{{- else }}
Forward the HTTP API to localhost (ClusterIP only):
  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "ophamin.fullname" . }}-http 8000:{{ .Values.service.http.port }}

Then point your client at http://127.0.0.1:8000 — e.g.:
  curl http://127.0.0.1:8000/health
  curl http://127.0.0.1:8000/openapi.json
  curl http://127.0.0.1:8000/docs    # Swagger UI
{{- end }}
{{- end }}
